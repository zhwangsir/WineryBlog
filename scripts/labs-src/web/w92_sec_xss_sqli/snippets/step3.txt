// 确实要渲染富文本：先转义，再交给 innerHTML
function escapeHtml(str) {
  return str.replace(/[&<>"']/g, (ch) => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' })[ch]);
}
box.innerHTML = `<p>${escapeHtml(userInput)}</p>`;
